cmake_minimum_required(VERSION 4.5)
project(FortranModules LANGUAGES NONE)
if(MOCK_FORTRAN)
  set(CMAKE_Fortran_COMPILER "${CMAKE_CURRENT_SOURCE_DIR}/mock-compiler.py")
  set(CMAKE_Fortran_COMPILER_ID GNU)
  set(CMAKE_Fortran_COMPILER_VERSION 14.2.0)
  set(CMAKE_Fortran_COMPILER_ID_RUN TRUE)
  set(CMAKE_Fortran_COMPILER_FORCED TRUE)
endif()
enable_language(Fortran)
try_compile(works SOURCE_FROM_CONTENT probe.f90 "program probe\nend program\n")
if(NOT works)
  message(FATAL_ERROR "The Fortran try_compile failed.")
endif()
add_executable(app main.f90 provider.f90)
set_target_properties(app PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
target_include_directories(app PRIVATE "${CMAKE_BINARY_DIR}/modules")
enable_testing()
add_test(NAME fortran_module COMMAND app)
