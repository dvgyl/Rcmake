cmake_minimum_required(VERSION 4.5)
project(AutogenConsumers LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
add_executable(autogen_app main.cpp widget.cpp widget.h panel.ui icons.qrc)
target_link_libraries(autogen_app PRIVATE Qt6::Widgets)
add_executable(autogen_other other.cpp)
target_link_libraries(autogen_other PRIVATE Qt6::Core)
enable_testing()
add_test(NAME autogen_app COMMAND autogen_app)
set_tests_properties(autogen_app PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
add_test(NAME autogen_other COMMAND autogen_other)
